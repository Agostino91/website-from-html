<script>
  // Switch nav color when scrolled off the hero image
  const nav = document.getElementById('nav');
  const hero = document.querySelector('.hero');
  function onScroll() {
    const heroBottom = hero.getBoundingClientRect().bottom;
    if (heroBottom < 70) nav.classList.add('over-light');
    else nav.classList.remove('over-light');
  }
  window.addEventListener('scroll', onScroll, { passive: true });
  onScroll();

  // Cycle language pill (demo)
  const langs = ['EN','IT','DE','FR'];
  let li = 0;
  const pill = document.getElementById('langPill');
  pill.addEventListener('click', () => {
    li = (li + 1) % langs.length;
    pill.textContent = langs[li];
  });
</script>
